SELECT d.dept_name, d.building, d.budget, 
  COUNT(*) OVER w_dept_count as db,
  AVG(d.budget) OVER w_building_avg as avg_building_budget,
  d.budget-AVG(d.budget) OVER w_building_avg_range as diff_from_nearby,
  PERCENT_RANK() OVER w_dept_order as budget_percentile
FROM department AS d
WINDOW w_base AS (PARTITION BY d.building),
       w_dept_count AS (w_base),
       w_building_avg AS (w_base),
       w_building_avg_range AS (w_base ORDER BY d.budget
                 RANGE BETWEEN 10000 PRECEDING AND 10000 FOLLOWING),
       w_dept_order AS (ORDER BY d.budget)
ORDER BY d.building, d.budget; 



#################################Slide 59#######################################
SELECT t.name, t.dept_name, t.salary, d.building,
    AVG(t.salary) OVER w_dept as avg_dept_salary,
    AVG(t.salary) OVER w_building as avg_building_salary,
    RANK() OVER w_dept_order as salary_rank_in_dept
FROM teacher t JOIN department d ON t.dept_name = d.dept_name
WINDOW w_dept AS (PARTITION BY t.dept_name),
        w_building AS (PARTITION BY d.building),
        w_dept_order AS (PARTITION BY t.dept_name 
                         ORDER BY t.salary DESC)
ORDER BY t.dept_name, t.salary DESC;

SELECT s.ID, s.name, t.year, t.semester, c.credits,
     SUM(c.credits) OVER w_running as running_total,
     AVG(c.credits) OVER w_moving as moving_avg_3_courses
FROM (student s JOIN takes t ON s.ID = t.ID) 
      JOIN course c ON t.course_id = c.course_id
WHERE t.grade IS NOT NULL AND t.grade != 'F'
WINDOW w_running AS (PARTITION BY s.ID 
                      ORDER BY t.year, t.semester),
        w_moving AS (w_running ROWS BETWEEN 2 PRECEDING AND CURRENT ROW)
ORDER BY s.ID, t.year, t.semester;
