#################################Slide 24#######################################
SELECT s.ID, s.name, s.dept_name, s.tot_cred, (CASE 
            WHEN s.tot_cred >= 100 THEN 'Senior'
            WHEN s.tot_cred >= 60 THEN 'Junior'
            WHEN s.tot_cred >= 30 THEN 'Sophomore'
            ELSE 'Freshman' END) AS standing,
      (CASE WHEN EXISTS (SELECT 1 FROM takes t WHERE t.ID = s.ID 
                                              AND (t.grade = 'F' OR t.grade IS NULL)) THEN 'At Risk'
             WHEN s.tot_cred < (SELECT AVG(tot_cred) FROM student) THEN 'Below Average Credits'
             ELSE 'Good Standing' END) AS status,
      (CASE WHEN (SELECT COUNT(*) FROM takes t WHERE t.ID = s.ID AND t.grade IN ('A', 'A-')) > 
                  (SELECT COUNT(*) FROM takes t WHERE t.ID = s.ID 
                                           AND t.grade IN ('C', 'C-', 'D', 'F')) THEN 'High Achiever'
             WHEN (SELECT COUNT(*) FROM takes t WHERE t.ID = s.ID 
                                           AND t.grade = 'F') > 2 THEN 'Academic Probation'
             ELSE 'Standard' END) AS category
FROM student s;
